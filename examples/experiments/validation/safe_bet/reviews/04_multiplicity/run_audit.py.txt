def clean(v):
    if isinstance(v,pd.DataFrame):return clean(v.to_dict('records'))
    if isinstance(v,pd.Series):return clean(v.to_list())
    if isinstance(v,np.ndarray):return clean(v.tolist())
    if isinstance(v,dict):return {str(k):clean(x) for k,x in v.items()}
    if isinstance(v,(list,tuple)):return [clean(x) for x in v]
    if isinstance(v,(pd.Timestamp,pd.Timedelta,Path)):return str(v)
    if isinstance(v,(np.integer,)):return int(v)
    if isinstance(v,(np.bool_,)):return bool(v)
    if isinstance(v,float) and not math.isfinite(v):return None
    if v is pd.NA:return None
    return v

def save(name,value):
    data[name]=value
    (out/(name+'.json')).write_text(json.dumps(clean(value),indent=2,allow_nan=False)+'\n')
    if isinstance(value,pd.DataFrame):value.to_csv(out/(name+'.csv'),index=False)

def check(name,actual,expected,atol=1e-12,rtol=1e-10):
    try: ok=bool(np.allclose(actual,expected,equal_nan=True,atol=atol,rtol=rtol))
    except Exception:ok=actual==expected
    checks.append({'check':name,'pass':ok,'actual':clean(actual),'expected':clean(expected)})
    return ok

def guarded(name,fn):
    try:
        result=fn();save(name,result);return result
    except Exception as e:
        caught.append({'case':name,'exception':type(e).__name__,'message':str(e)})
        return None

# Public data construction: every assignment has an anchor, conversion events follow.
def binary_stream(strata_counts,second_metric=False,dates=None):
    rows=[];assign=[]
    for st,arms in strata_counts.items():
        for arm,(n,k) in arms.items():
            for i in range(n):
                uid=f'{st}_{arm}_{i}';assign.append((uid,arm,pd.Timestamp('2026-01-01'),st))
                ds=dates or ['2026-01-02']
                for dayno,day in enumerate(ds):
                    rows.append((uid,'anchor',pd.Timestamp(day),arm,st,1.))
                    kk=k if dayno==0 else max(1,k-5)
                    if i<kk:rows.append((uid,'conversion',pd.Timestamp(day)+pd.Timedelta(hours=1),arm,st,1.))
                    if second_metric and i>=n-kk:rows.append((uid,'secondary',pd.Timestamp(day)+pd.Timedelta(hours=2),arm,st,1.))
    events=pd.DataFrame(rows,columns=['user_id','event','timestamp','arm','stratum','amount'])
    roster=pd.DataFrame(assign,columns=['user_id','arm','assigned_at','stratum']).set_index('user_id')
    stream=Eventstream(events,{'path_cols':['user_id'],'segment_cols':['arm','stratum']})
    return stream,roster,events
CONV={'name':'conversion','metric':'has_event','metric_args':{'event':'conversion'}}
ANCHOR={'name':'always','metric':'has_event','metric_args':{'event':'anchor'}}
DUP={'name':'conversion_duplicate','metric':'has_event','metric_args':{'event':'conversion'}}
SECOND={'name':'secondary','metric':'has_event','metric_args':{'event':'secondary'}}
DIFF=('arm','treatment','control')
def score(stream,metrics,**kw):
    kw.setdefault('min_support',1)
    return stream.get_ab_test(DIFF,metrics=metrics,**kw)

# Actual small-n Welch power under the advertised normal-approximation MDE.
power_records=[]
power_rng=np.random.default_rng(4072027)
for n in [3,5,10,30,100]:
    delta=oracle.approximate_mde(math.sqrt(2/n),.05,.8)
    # independent normal arms with known unit variance, Welch test estimates it.
    a=power_rng.normal(0,1,(R,n));b=power_rng.normal(delta,1,(R,n))
    se2=a.var(axis=1,ddof=1)/n+b.var(axis=1,ddof=1)/n
    denom=(a.var(axis=1,ddof=1)/n)**2/(n-1)+(b.var(axis=1,ddof=1)/n)**2/(n-1)
    dof=se2**2/denom;t=(b.mean(axis=1)-a.mean(axis=1))/np.sqrt(se2)
    rej=2*stats.t.sf(abs(t),dof)<.05;k=int(rej.sum());rate=k/R;se=math.sqrt(rate*(1-rate)/R)
    power_records.append({'seed':4072027,'n_per_arm':n,'replications':R,'delta':delta,'nominal_power':.8,
        'rejections':k,'power':rate,'mc_se':se,'normal_mc_ci_low':rate-1.96*se,'normal_mc_ci_high':rate+1.96*se})
